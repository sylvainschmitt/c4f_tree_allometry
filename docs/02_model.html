Model

\[ \begin{align} log(agb) \sim \mathcal N (log(\mu), \sigma) \\ \mu = \sum_{i \in p} \gamma_s \times 0.0673 \times [wd_i \times dbh_i^2 \times \frac{\alpha_f \times dbh_i}{\beta_f+dbh_i}]^{0.976} \\ \gamma_s = exp(\epsilon_s \times \sigma_s) \\ \alpha_f \sim \mathcal N (\beta_0,\sigma_a) \\ \beta_f \sim \mathcal N (\beta_0,\sigma_b) \\ \epsilon_s \sim \mathcal N (0,1) \\ \end{align} \]

With plot biomass \(agb\) linked to the sum of tree biomass \(\mu\) defined on tree wood density \(wd_i\), tree diameter \(dbh_i\) and tree height \(h_i\) itself derived from \(dbh_i\) following a Michaelis Menten law with parameters \(\alpha_f\) and \(\beta_f\) depending on the factor \(f\) being the intersection of the \(system\) and the \(origin\) and a random parameters \(\gamma_s\) representing species random effect.

trees <- read_csv2("data/all_tree_corr.csv") %>% 
  # filter(database == "Amani") %>% # test
  select(-1) %>% 
  mutate(system = gsub("_", " ", system)) %>% 
  select(database, id_plot, id_tree, system, origin, dbh, H, genusCorr, speciesCorr) %>% 
  rename(h = H, genus = genusCorr, species = speciesCorr) %>% 
  filter(!is.na(dbh), !is.na(h)) %>% 
  mutate(factornum = as.numeric(as.factor(paste(system, origin)))) %>% 
  mutate(speciesnum = as.numeric(as.factor(paste(genus, species)))) %>% 
  mutate(plotnum = as.numeric(as.factor(id_plot))) %>% 
  arrange(plotnum, speciesnum, factornum) %>% 
  mutate(obs = 1:n())
trees$wd <- BIOMASS::getWoodDensity(genus = trees$genus,
                                    species = trees$species,
                                    stand = trees$id_plot)$meanWD
plots <- trees %>% 
  mutate(agb = 0.0673*(wd*dbh^2*h)^0.976) %>% 
  group_by(database, system, id_plot, plotnum) %>% 
  summarise(agb = sum(agb)) 
positions <- trees %>% 
  group_by(plotnum) %>% 
  summarise(start = min(obs), end = max(obs))
mdata <- list(
  N = nrow(trees),
  P = nrow(plots),
  S = max(trees$speciesnum),
  "F" = max(trees$factornum),
  plot = trees$plotnum,
  species = trees$speciesnum,
  factor = trees$factornum,
  agb = plots$agb,
  dbh = trees$dbh,
  wd = trees$wd,
  start = positions$start,
  end = positions$end
)

model <- cmdstan_model("models/factor.stan")
path <- "chains/systemorigin"
if(file.exists(path))
  unlink(path, recursive = TRUE)
dir.create(path)
fit <- model$sample(
    data = mdata,
    chains = 4,
    parallel_chains = 4,
    save_warmup = FALSE,
    output_dir = path,
    refresh = 100
  )

            quantiles = ~ posterior::quantile2(., probs = c(0.05, 0.25, 
                                                            0.5, 0.75, 0.95))) %>%

  select(-f) %>% 
  left_join(select(trees, system, origin, factornum) %>% unique()) %>% 
  select(-factornum) %>% 
  gather(quantile, value, -parameter, -system, -origin) %>% 
  mutate(system = gsub("_", " ", system)) %>%

coords <- read_csv2("data/all_plot_corr.csv") %>% 
  select(database, id_plot, long, lat)

                                       WD = trees$wd, 
                                       coord = left_join(trees, coords)[c("long", "lat")])
agb_chave <- trees %>% 
  group_by(database, system, id_plot, plotnum) %>% 
  summarise(chave = sum(agb_chave)*10^3) 
plots %>% 
  left_join(agb_chave) %>% 
  ungroup() %>% 
  mutate(model = fit$summary("pred", "median")$median) %>% 
  gather(model, predicted, -database, -system, -id_plot, -plotnum, -agb) %>% 
  rename(observed = agb) %>% 
  mutate(error_log = log(observed) - log(predicted)) %>% 
  ggplot(aes(system, error_log, fill = model)) +
  geom_boxplot() +
  theme_bw() +
  coord_flip()
